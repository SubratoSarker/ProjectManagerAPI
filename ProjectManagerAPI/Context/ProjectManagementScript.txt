CREATE FUNCTION [dbo].[funAgging]
(
	@Date DATE
)
RETURNS VARCHAR(100)
AS
BEGIN
	-- Default to today if @Date is NULL
	SET @Date = ISNULL(@Date, GETDATE());

	DECLARE @Years INT = DATEDIFF(YEAR, @Date, GETDATE());
	DECLARE @Months INT = DATEDIFF(MONTH, DATEADD(YEAR, @Years, @Date), GETDATE());
	DECLARE @Days INT = DATEDIFF(DAY, DATEADD(MONTH, @Months, DATEADD(YEAR, @Years, @Date)), GETDATE());

	IF @Days < 0
	BEGIN
		SET @Months = @Months - 1;
		SET @Days = DATEDIFF(DAY, DATEADD(MONTH, @Months, DATEADD(YEAR, @Years, @Date)), GETDATE());
	END

	IF @Months < 0
	BEGIN
		SET @Years = @Years - 1;
		SET @Months = DATEDIFF(MONTH, DATEADD(YEAR, @Years, @Date), GETDATE());
		SET @Days = DATEDIFF(DAY, DATEADD(MONTH, @Months, DATEADD(YEAR, @Years, @Date)), GETDATE());
	END

	DECLARE @Length VARCHAR(100) = '';

	IF @Years > 0
		SET @Length = CONCAT(@Length, @Years, ' year', CASE WHEN @Years != 1 THEN 's ' ELSE ' ' END);

	IF @Months > 0
		SET @Length = CONCAT(@Length, @Months, ' month', CASE WHEN @Months != 1 THEN 's ' ELSE ' ' END);

	IF @Days > 0
		SET @Length = CONCAT(@Length, @Days, ' day', CASE WHEN @Days != 1 THEN 's' ELSE '' END);

	IF @Length = ''
		SET @Length = 'Today';

	SET @Length = RTRIM(@Length);

	RETURN @Length;
END

-- Description:	<Description, ,>
-- =============================================
CREATE FUNCTION [dbo].[funTaskPerProjectByUser]
(
	@Project int,
	@Enroll int
)
RETURNS int
AS
BEGIN
	Declare @Count int
	set @Count=(SELECT isNull(count(*),0)
		  FROM ProjectManagement.dbo.tblTask t
		  inner join ProjectManagement.dbo.tblTaskDetail d on t.intTaskID=d.intTaskID
		  where d.intUserID=@Enroll and t.intProjectID=@Project and d.intStatusID not in (5,6)
		  )
	return @count
END

-- =============================================
-- Author:		<Author,,Name>
-- Create date: <Create Date,,>
-- Description:	<Description,,>
-- EXEC PerformanceReprot '2024-6-1', '2024-9-1', 2, 0
-- =============================================
CREATE PROCEDURE [dbo].[PerformanceReprot]
	@FromDate date,
	@ToDate date,
	@User int,
	@Project int
AS
BEGIN
	WITH CTE_ConvertedTime AS (
	SELECT
		intUserID,
		intTaskID,
		CAST(dteInsert AS DATE) AS dteInsertDate,
		DATEDIFF(SECOND, '00:00:00', tmWorking) AS WorkingSeconds
	FROM
		ProjectManagement.dbo.tblWorkLog
	WHERE
		CAST(dteInsert AS DATE) BETWEEN @FromDate AND @ToDate
	),
	CTE_SumTime AS (
		SELECT
			intUserID,
			intTaskID,
			SUM(WorkingSeconds) AS TotalWorkingSeconds
		FROM
			CTE_ConvertedTime
		GROUP BY
			intUserID,
			intTaskID
	),
	CTE_TotalTime AS (
		SELECT
			SUM(TotalWorkingSeconds) AS TotalWorkingSecondsAll
		FROM
			CTE_SumTime
		WHERE
			intUserID = @User
	)
	SELECT 
		t.intTaskID,
		t.strTaskName,
		t.strDescription,
		t.strRequestFrom,
		p.strProjectName,
		CONVERT(TIME, DATEADD(SECOND, st.TotalWorkingSeconds, '00:00:00')) AS tmWorking,
		CAST(ROUND(100.0 * st.TotalWorkingSeconds / tt.TotalWorkingSecondsAll, 2) AS DECIMAL(5, 2)) AS WorkPercent
	FROM 
		ProjectManagement.dbo.tblTask t 
		INNER JOIN ProjectManagement.dbo.tblProject p ON p.intProjectID = t.intProjectID
		INNER JOIN CTE_SumTime st ON st.intTaskID = t.intTaskID 
		CROSS JOIN CTE_TotalTime tt
	WHERE 
		st.intUserID=@User and p.intProjectID = case when @Project=0 then p.intProjectID else @Project end
END

-- Description:	<Description,,>
-- exec PerformanceReprotTeam 2, '2024-01-01', '2024-10-01'
-- =============================================
CREATE PROCEDURE [dbo].[PerformanceReprotTeam]
	@User int,
	@FromDate date,
	@ToDate date
AS
BEGIN

	WITH CTE_ConvertedTime AS (
		SELECT
			intUserID,
			CAST(dteInsert AS DATE) AS dteInsertDate,
			DATEDIFF(SECOND, '00:00:00', tmWorking) AS WorkingSeconds
		FROM dbo.tblWorkLog
		WHERE CAST(dteInsert AS DATE) BETWEEN @FromDate AND @ToDate
	),
	CTE_SumTime AS (
		SELECT
			intUserID,
			dteInsertDate,
			SUM(WorkingSeconds) AS TotalWorkingSeconds
		FROM CTE_ConvertedTime
		GROUP BY intUserID, dteInsertDate
	)
	select cast( ROW_NUMBER() OVER (ORDER BY dteInsertDate , u.intUserID) as int) as ID, cast(u.intUserID as int) as intUserID, u.strUserName , t.dteInsertDate, CONVERT(TIME, DATEADD(SECOND, t.TotalWorkingSeconds, '00:00:00')) AS tmWorking
	from CTE_SumTime t
	inner join ProjectManagement.dbo.tblUser u on u.intUserID=t.intUserID
	inner join ProjectManagement.dbo.tblTeam tm on tm.intTeamID=u.intTeamID
	where tm.intUserID=@User
END

CREATE PROCEDURE [dbo].[sprAssignUser]
	@User int,
	@Project int,
	@Active bit,
	@Enroll int,
	@Msg varchar(MAX) out
AS
BEGIN
	Declare @Details varchar(max)
	Declare @UserName varchar(MAX), @ProjectName varchar(MAX),  @Notify varchar(MAX)
	select top 1 @UserName=strUserName from ProjectManagement.dbo.tblUser where intUserID=@User
	select top 1 @ProjectName=strProjectName from ProjectManagement.dbo.tblProject where intProjectID=@Project

	IF Exists(select * from ProjectManagement.dbo.tblProjectAssign where intUserID=@User and intProjectID=@Project)
	Begin
		Declare @ActiveState bit= (select top 1 isActive from ProjectManagement.dbo.tblProjectAssign where intUserID=@User and intProjectID=@Project)
		IF(@ActiveState!=@Active)
		Begin
			update ProjectManagement.dbo.tblProjectAssign set isActive=@Active where intUserID=@User and intProjectID=@Project
			IF(@Active = 0)
			Begin 
				set @Details='User -'+cast( @User as Varchar(50))+' Removed from Project -'+cast( @Project as Varchar(50))
				EXEC ProjectManagement.dbo.sprLog 'Update', 'Project Assign', @Details, @Enroll
				set @Msg=@UserName+' Removed form '+@ProjectName+' Successfully.'
				set @Notify ='You have been Removed from '+@ProjectName
				exec dbo.sprNotify @User, @Notify
			End
			Else
			Begin
				set @Details='User -'+cast( @User as Varchar(50))+' ReAssign To Project -'+cast( @Project as Varchar(50))
				EXEC ProjectManagement.dbo.sprLog 'Update', 'Project Assign', @Details, @Enroll
				set @Msg=@UserName+' ReAssign To '+@ProjectName+' Successfully.'
				set @Notify ='You have been ReAssign To '+@ProjectName
				exec dbo.sprNotify @User, @Notify
			End
		End
		else IF(@ActiveState=@Active)
		Begin
			set @Msg='Unprocess.'
		End
	End
	else IF not Exists(select * from ProjectManagement.dbo.tblProjectAssign where intUserID=@User and intProjectID=@Project)
	Begin
		IF(@Active=1)
		Begin
			insert into ProjectManagement.dbo.tblProjectAssign (intUserID, intProjectID) 
			values (@User, @Project)
			set @Details='User -'+cast( @User as Varchar(50))+' Assign To Project -'+cast( @Project as Varchar(50))
			EXEC ProjectManagement.dbo.sprLog 'Insert', 'Project Assign', @Details, @Enroll
			set @Msg=@UserName+' Assign To '+@ProjectName+' Successfully.'
			set @Notify ='You have been Assign To '+@ProjectName
			exec dbo.sprNotify @User, @Notify
		End
		Else
		Begin
			set @Msg='Unprocess.' 
		End
	End
END

-- =============================================
-- Author:		<Author,,Name>
-- Create date: <Create Date,,>
-- Description:	<Description,,>
-- exec dbo.sprPendingReport 2
-- =============================================
CREATE PROCEDURE [dbo].[sprPendingReport]
	@User int
AS
BEGIN
	WITH CTE_ConvertedTime AS (
		SELECT
			intUserID,
			intTaskID,
			CAST(dteInsert AS DATE) AS dteInsertDate,
			DATEDIFF(SECOND, '00:00:00', tmWorking) AS WorkingSeconds
		FROM
			ProjectManagement.dbo.tblWorkLog
		),
		CTE_SumTime AS (
			SELECT
				intUserID,
				intTaskID,
				SUM(WorkingSeconds) AS TotalWorkingSeconds
			FROM
				CTE_ConvertedTime
			GROUP BY
				intUserID,
				intTaskID
		)
	select t.intTaskID, p.strProjectName, t.strTaskName, t.strDescription, s.strStatus, CONVERT(TIME, DATEADD(SECOND, sp.TotalWorkingSeconds, '00:00:00')) AS tmWorking
	from ProjectManagement.dbo.tblTask t 
	inner join ProjectManagement.dbo.tblTaskDetail td on td.intTaskID=t.intTaskID
	inner join ProjectManagement.dbo.tblStatus s on s.intStatusID=td.intStatusID
	inner join ProjectManagement.dbo.tblProject p on p.intProjectID=t.intProjectID
	inner join CTE_SumTime sp on sp.intTaskID=t.intTaskID and sp.intUserID=td.intUserID
	where td.intUserID=@User and s.intStatusID in (1,2,3,4)
END

CREATE PROCEDURE [dbo].[sprSecurityCheck] 
	@type int,
	@Email varchar(200),
	@password varchar(200),
	@code Varchar(200)
AS
BEGIN
if(@Email like '%_@__%.__%')
begin
	Begin try
	Begin Transaction
	DECLARE @Return TABLE (
		UserID INT,
		UserName VARCHAR(255),
		Phone VARCHAR(20),
		Email VARCHAR(255),
		TeamID INT,
		TeamName VARCHAR(255),
		ISBoss BIT,
		ISAdmin BIT,
		Token VARCHAR(255),
		Response VARCHAR(255));
	declare @count int,@UserID int
	if exists( select 1 from ProjectManagement.dbo.tblUser where strEmail= @Email) 
	Begin 
		set @count= (select count(*) from ProjectManagement.dbo.tblUser where strEmail= @Email)
		IF(@count!=1)
		Begin
			insert into @Return values(0,'Name','Phone','Email',0,'Team',0,0,'','Invalid Credantial')
		End
		else
		Begin
			Set @UserID=(select top 1 intUserID from ProjectManagement.dbo.tblUser where strEmail= @Email)
			IF exists(select 1 from ProjectManagement.dbo.tblUser where intUserID= @UserID and isActive=0)
			Begin
				insert into @Return values(0,'Name','Phone','Email',0,'Team',0,0,'','Sorry, you are no longer authorized to access this service')
			End
			Else 
			Begin
				IF(@type=0) --LogIn
			Begin
				IF((select top 1 dteValidTo from ProjectManagement.dbo.tblUserSecurity where intUserID=@UserID)<GETDATE())
				BEGIN
					insert into @Return values(0,'Name','Phone','Email',0,'Team',0,0,'','Password Expaired !!!')
				END
				ELSE IF((select top 1 isLocked from ProjectManagement.dbo.tblUserSecurity where intUserID=@UserID)=1)
				BEGIN
					insert into @Return values(0,'Name','Phone','Email',0,'Team',0,0,'','Your Account is locked !!!')
				END
				Else
				Begin
					Declare @Userpass varbinary(max)= HASHBYTES('SHA2_512',@password+cast('{Task}' as varchar(20)) )
					Declare @DBpass varbinary(max) = isnull((select top (1) strKey from ProjectManagement.dbo.tblUserSecurity where intUserID = @UserID ), cast('NDOTP' as varbinary))
					IF(@Userpass!=@DBpass)
					Begin 
						Update  ProjectManagement.dbo.tblUserSecurity set intFailedAttamped=intFailedAttamped+1 where intUserID=@UserID
						insert into @Return values(0,'Name','Phone','Email',0,'Team',0,0,'','Invalid Credantial')
					End
					Else
					Begin
						insert into @Return 
						select u.intUserID, u.strUserName, u.strPhone, u.strEmail, t.intTeamID, t.strTeamName, case when t.intUserID=u.intUserID then 1 else 0 end, u.isAdmin, '', 'GO'
						from ProjectManagement.dbo.tblUser u
						inner join ProjectManagement.dbo.tblTeam t on t.intTeamID=u.intTeamID
						where u.intUserID=@UserID
					End
				End
			End
			Else IF(@type=1)
			Begin
				DECLARE @Key VARCHAR(50)
				SET @Key = ''
				SELECT TOP 6 @Key = @Key + [char]
					FROM [dbo].[tblRandomCharsHelper]
					ORDER BY NEWID()
				Insert Into ProjectManagement.dbo.tblSecurityCheck (intUserID,strOTP) values(@UserID,@Key)
				insert into @Return values(0,'Name','Phone','Email',0,'Team',0,0,'','Otp send to :'+(select top 1 strPhone from ProjectManagement.dbo.tblUser where intUserID= @UserID))
				EXEC ProjectManagement.dbo.sprLog 'Insert', 'OTP Request', 'OTP Request', @UserID
			End
			Else IF(@type=2)
			Begin
				Declare @OTP varchar(20),@intOtpId int
				select top 1 @OTP=strOTP, @intOtpId=intId from  ProjectManagement.dbo.tblSecurityCheck where intUserID=@UserID and GETDATE() between dteInsertTime and dteExpTime and isUsed=0 order by intId desc
				IF(@OTP=@code)
				Begin
					Declare @pass varbinary(max)= HASHBYTES('SHA2_512',@password+cast('{Task}' as varchar(20)) )
					Update ProjectManagement.dbo.tblUserSecurity set strKey=@pass, dteValidTo=DATEADD(MONTH,1,GETDATE()) where intUserID=@UserID
					Update ProjectManagement.dbo.tblSecurityCheck set isUsed=1 where intId=@intOtpId
					EXEC ProjectManagement.dbo.sprLog 'Update', 'Password Reset', 'Password Reset Successfully', @UserID
					insert into @Return values(0,'Name','Phone','Email',0,'Team',0,0,'','Password Updated Successfully')
				End
				Else
				Begin
					insert into @Return values(0,'Name','Phone','Email',0,'Team',0,0,'','Otp dose not Match !!!')
				End
			End
			END
		End
	End
	Else
	Begin
		insert into @Return values(0,'Name','Phone','Email',0,'Team',0,0,'','Invalid Credantial')
	End
	Commit
	select * from @Return
	End try
	Begin Catch
		Rollback
		Insert into @Return (Response) Values('DB_Error:'+ ERROR_MESSAGE())
		select * from @Return
	End Catch
END
END
